Cine-AI-Scene: Intelligent Cinema & Real-Time Seat Reservation

Category: AI & Real-Time Web Application • Stack: React 18, TypeScript, Vite, Supabase Realtime, Tailwind CSS

An AI-powered cinema discovery and interactive seat reservation platform. Features intelligent AI movie hooks and audience sentiment summaries, dynamic theater screening schedules, and a real-time reactive seating matrix with row pricing tiers powered by Supabase PostgreSQL Realtime channels.

Core Architecture & Deliverables:
  • AI Movie Hooks & Recommendations: Generative AI storylines and predictive viewer-fit ratings for featured releases
  • Real-Time Seat Booking Engine: Interactive seat layout (VIP, Premium, Standard) with real-time selection and pricing
  • PostgreSQL Realtime Subscriptions: Instant concurrency locks via supabase.channel('seats-{screeningId}') to prevent double booking
  • Modern Frontend Stack: Built with Vite, React 18, TypeScript, Tailwind CSS, shadcn/ui, and TanStack React Query
// Cine-AI-Scene: Real-time Seat Subscription with Supabase Channels
import { useEffect } from "react";
import { supabase } from "@/integrations/supabase/client";
import { useQueryClient } from "@tanstack/react-query";

export const useRealtimeSeatSync = (screeningId: string) => {
  const queryClient = useQueryClient();

  useEffect(() => {
    if (!screeningId) return;

    // Listen to real-time seat reservation changes across clients
    const channel = supabase
      .channel(`seats-${screeningId}`)
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'seats',
          filter: `screening_id=eq.${screeningId}`
        },
        () => {
          // Invalidate cache and update seat grid instantly
          queryClient.invalidateQueries({ queryKey: ["seats", screeningId] });
        }
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [screeningId, queryClient]);
};

// Cine-Genius-Booking: Screening Dispatcher & Theaters Query
import { useQuery } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";
import type { Movie, Screening } from "@/types/database";

export const useUpcomingScreenings = (movieId: string) => {
  return useQuery({
    queryKey: ['screenings', movieId],
    queryFn: async () => {
      const today = new Date().toISOString().split('T')[0];
      const { data, error } = await supabase
        .from('screenings')
        .select('*, theatres(*)')
        .eq('movie_id', movieId)
        .gte('show_date', today)
        .order('show_date', { ascending: true })
        .order('show_time', { ascending: true })
        .limit(10);
        
      if (error) throw error;
      return data as Screening[];
    },
  });
};

// Scoop-Sweet-Suite: Real-time Live Catalog Subscription
import { useEffect, useState } from "react";
import { supabase } from "@/integrations/supabase/client";

export const useRealtimeCatalog = () => {
  const [iceCreams, setIceCreams] = useState([]);

  useEffect(() => {
    fetchIceCreams();

    // Subscribe to database changes to reflect live stock levels
    const channel = supabase
      .channel("icecreams-catalog-sync")
      .on(
        "postgres_changes",
        { event: "*", schema: "public", table: "icecreams" },
        () => {
          // Re-fetch automatically when inventory or stock updates
          fetchIceCreams();
        }
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, []);

  const fetchIceCreams = async () => {
    const { data } = await supabase.from("icecreams").select("*").order("name");
    if (data) setIceCreams(data);
  };

  return { iceCreams };
};
